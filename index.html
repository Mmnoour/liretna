function calculateChange() { let newCash = parseFloat(document.getElementById('newCash').value) || 0; let oldCash = parseFloat(document.getElementById('oldCash').value) || 0; let priceInput = parseFloat(document.getElementById('price').value) || 0; // قراءة نوع العملة للسعر (جديدة أو قديمة) let priceType = document.querySelector('input[name="price_type"]:checked')?.value || 'new'; // تحويل السعر للجديد: إذا كان السعر بالقديم نقسمه على 100 let priceInNew = priceType === 'old' ? priceInput / 100 : priceInput; // إجمالي المحفظة بالجديد: القديم يُقسم على 100 (حذف الصفرين) let totalInNew = newCash + (oldCash / 100); let resultBox = document.getElementById('resultBox'); resultBox.style.display = 'block'; document.getElementById('totalWallet').innerText = `إجمالي ما تملك (بالعملة الجديدة): ${totalInNew} ليرة.`; if (totalInNew < priceInNew) { document.getElementById('status').innerText = "❌ عذراً، المبلغ اللي معك ما بيكفي للشراء!"; document.getElementById('changeNew').innerText = ""; document.getElementById('changeOld').innerText = ""; return; } document.getElementById('status').innerText = "✅ المبلغ كافي!"; // الباقي الإجمالي بالعملة الجديدة let remainingInNew = totalInNew - priceInNew; // معالجة الفكة: سحب الفئات الصحيحة بالجديد (مضاعفات الـ 10 ليرات جديدة) let changeNewPart = Math.floor(remainingInNew / 10) * 10; let fraction = remainingInNew - changeNewPart; // الكسور (أقل من 10 ليرات جديدة) // تحويل الفراطة الباقية إلى عملة قديمة بضربها بـ 100 let changeOldPart = Math.round(fraction * 100); document.getElementById('changeNew').innerText = `الباقي بالعملة الجديدة: ${changeNewPart} ليرة.`; if (changeOldPart > 0) { document.getElementById('changeOld').innerText = `🪙 الفراطة بالعملة القديمة: ${changeOldPart} ليرة.`; } else { document.getElementById('changeOld').innerText = ""; } }
